escann_knights_reform = {
	icon = "manchu_warrior"
	allow_normal_conversion = yes

	potential = {
		OR = {
			tag = B60
			tag = B58
			tag = B32
			tag = B33
		}
	}

	trigger = {
		if = {
			limit = { has_modifier_locking_military_tier = yes }
			OR = {
				has_reform = escann_knights_reform
				has_not_modifier_locking_military_tier = yes
			}
		}
	}

	modifiers = {
		army_tradition_decay = -0.005
		movement_speed = 0.1
	}

	# 解锁骑士征召：只有挂了 escann_knights_location 的省份才计入
	custom_attributes = {
		can_recruit_banners_from_primary_culture_provinces = yes
	}

	effect = {
		hidden_effect = {
			# 首都额外 +10 队骑士
			capital_scope = {
				add_province_modifier = {
					name = escann_knights_capital
					duration = -1
					hidden = yes
				}
			}
			# 东坎诺五文化省份标记为骑士征召地
			every_owned_province = {
				limit = { escann_knights_culture = yes }
				add_province_modifier = {
					name = escann_knights_location
					duration = -1
					hidden = yes
				}
			}
		}
	}

	removed_effect = {
		hidden_effect = {
			capital_scope = {
				remove_province_modifier = escann_knights_capital
			}
			every_owned_province = {
				limit = { has_province_modifier = escann_knights_location }
				remove_province_modifier = escann_knights_location
			}
		}
	}

	ai = {
		factor = 100
	}
}
